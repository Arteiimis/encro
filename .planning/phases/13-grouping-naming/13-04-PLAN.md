    ```cpp
    TEST_CASE(
      "execute() with picture naming produces subPart split for size overflow",
      "[picture-process][pack]"
    ) {
      TempDir temp;
      auto const inputDir = temp.path / "pics";
      auto const outputDir = temp.path / "packed";
      fs::create_directories(inputDir);

      constexpr auto kPictureSize = std::uintmax_t{240ULL * 1024ULL * 1024ULL};
      auto const f1 = createSparseSizedFile(inputDir, "a.jpg", kPictureSize);
      auto const f2 = createSparseSizedFile(inputDir, "b.jpg", kPictureSize);
      auto const f3 = createSparseSizedFile(inputDir, "c.jpg", kPictureSize);

      auto config = appctx::AppConfig{};
      config.processType = "picture";
      config.recursive = true;
      config.yesToAll = true;
      config.pictureFolderSummary = true;
      config.outputLayout = appctx::OutputLayout::Flat;
      config.inputPath = inputDir;

      auto const scannedPics = std::vector<fs::path>{f1, f2, f3};
      auto const plannedEntryNames = planPictureZipEntryNames(config, inputDir, scannedPics);
      auto const packInputs = buildPicturePackEntryInputs(
        config, inputDir, scannedPics, plannedEntryNames);

      auto sortedEntries = std::vector<fs::path>{};
      sortedEntries.reserve(packInputs.size());
      for (auto const& input : packInputs) { sortedEntries.push_back(input.entry.sourcePath); }

      auto entryNameForFile = [plannedEntryNames = std::move(plannedEntryNames)](
        fs::path const& filePath) {
        auto const it = plannedEntryNames.find(filePath);
        return it != plannedEntryNames.end()
          ? it->second
          : filePath.filename().generic_string();
      };

      auto const result = pack::execute(pack::PackRequest{
        .entries = std::move(sortedEntries),
        .mode = pack::PackMode::Media,
        .outputDir = outputDir,
        .compact = true,
        .removeOnFailure = true,
        .naming = pack::NamingConfig{
          .layout = appctx::OutputLayout::Flat,
          .forceConflictHandling = false,
          .baseName = "pics",
        },
        .entryNameForFile = std::move(entryNameForFile),
      });

      REQUIRE(result.has_value());
      REQUIRE(result->exitCode == 0);

      // 3 files at 240MB each = 720MB > 500MB limit → should produce 2 zip files (subPart split)
      REQUIRE(result->zippedFiles.size() == 2);

      // Zip file names should include subPart suffix
      auto const zip1Name = result->zippedFiles[0].filename().string();
      auto const zip2Name = result->zippedFiles[1].filename().string();
      CHECK(zip1Name.find("pics_part1.1") != std::string::npos);
      CHECK(zip2Name.find("pics_part1.2") != std::string::npos);
    }
    ```
