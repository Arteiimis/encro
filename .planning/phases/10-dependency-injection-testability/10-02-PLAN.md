After `using enum terminal::MessageKind;` (line 33), add:

```cpp
namespace {

struct ZipWriter {
    libzippp::ZipArchive zip;
    bool opened = false;

    explicit ZipWriter(const std::string& path) : zip(path) {}

    void open() {
        zip.open(libzippp::ZipArchive::New);
        opened = true;
    }

    ~ZipWriter() {
        if (opened) {
            try {
                zip.close();
            } catch (...) {
            }
        }
    }

    ZipWriter(const ZipWriter&) = delete;
    ZipWriter& operator=(const ZipWriter&) = delete;
};

}  // namespace
```
