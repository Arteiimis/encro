    Test PackService's interaction with IPacker interface:
    1. packGroups delegates to IPacker::packFilesToZip for each group
    2. packGroups passes correct zip file paths derived from plan outputDir + zipNameForIndex
    3. packGroups propagates errors from IPacker
    4. packGroups returns correct zippedFile paths on success
    5. packGroups handles empty plan (zero groups)
    6. packGroups compact mode calls compact overload (isCompact=true)
    7. packGroups full-progress mode calls full overload (isCompact=false)
    8. packAllFilesInDirectory delegates to buildDirectoryPackPlan then packGroups
    9. buildDirectoryPackPlan receives correct parameters from packAllFilesInDirectory
    10. Error from buildDirectoryPackPlan propagates through packAllFilesInDirectory
  </behavior>
  <implementation>
    Tests follow Arrange-Act-Assert pattern:
    - Arrange: Create MockPacker, inject into PackService, set up PackPlan
    - Act: Call PackService method
    - Assert: Inspect MockPacker capture vectors for expected calls
    
    Key conventions:
    - Each test case creates a fresh MockPacker and PackService
    - Use `pack::test::MockPacker` from tests/packer_mock.h
    - Use `pack::PackPlan` with minimal groups (1-2 entries per group)
    - No TempDir needed — no filesystem access
    - Use Catch2 SECTIONs for related scenarios
  </implementation>

#include "pack/pack_service.h"
#include "pack/pack_types.h"
#include "packer_mock.h"

#include <catch2/catch_all.hpp>

#include <filesystem>
#include <memory>
#include <vector>

namespace fs = std::filesystem;

using namespace pack::test;

TEST_CASE("packGroups delegates to IPacker::packFilesToZip for each group", "[pack-service][mock]") {
    auto mock = std::make_unique<MockPacker>();
    auto& mockRef = *mock;
    PackService service(std::move(mock));

    PackPlan plan{
        .groups = {
            {PackFileEntry{.sourcePath = "a.txt", .zipEntryName = "a.txt"}},
            {PackFileEntry{.sourcePath = "b.txt", .zipEntryName = "b.txt"}},
        },
        .outputDir = "/out",
        .zipNameForIndex = [](std::size_t i) { return std::format("part{}.zip", i + 1); },
    };

    auto result = service.packGroups(plan);

    REQUIRE(result);
    REQUIRE(mockRef.packFilesToZipCalls.size() == 2);
    CHECK(mockRef.packFilesToZipCalls[0].entries.size() == 1);
    CHECK(mockRef.packFilesToZipCalls[1].entries.size() == 1);
}

TEST_CASE("packGroups passes correct zip paths", "[pack-service][mock]") {
